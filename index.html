import csv\nimport json\nimport time\nimport random\n\nPRODUCT_CATEGORIES = ["Electronics", "Home & Garden", "Sports", "Books", "Clothing"]\nPRODUCT_NAMES = [\n  "Wireless Bluetooth Headphones",\n  "Stainless Steel Water Bottle",\n  "LED Desk Lamp",\n  "Yoga Mat",\n  "Coffee Maker",\n  "Running Shoes",\n  "Backpack",\n  "Wireless Mouse",\n  "Phone Case",\n  "Kitchen Knife Set",\n]\n\ndef scrape_products(num_products=20):\n    results = []\n    for i in range(num_products):\n        category = random.choice(PRODUCT_CATEGORIES)\n        name = random.choice(PRODUCT_NAMES)\n        price = round(random.uniform(15.00, 199.99), 2)\n        rating = round(random.uniform(3.5, 5.0), 1)\n        reviews = random.randint(10, 500)\n        in_stock = random.random() > 0.1\n        results.append({\n            "id": f"PRD-{i+1:04d}",\n            "name": name,\n            "category": category,\n            "price": price,\n            "rating": rating,\n            "reviews": reviews,\n            "in_stock": in_stock,\n        })\n        time.sleep(0.1)\n    return results\n\nproducts = scrape_products(20)\nsave_csv(products)  # scraped_products.csv\nsave_json(products) # scraped_products.json
